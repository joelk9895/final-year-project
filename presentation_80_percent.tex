\begin{frame}{Project Overview}
    \begin{itemize}
        \item \textbf{Objective}: Develop a seamless, semi-autonomous EV charging station authorization system.
        \item \textbf{Method}: Automated License Plate Recognition (ALPR) via computer vision.
        \item \textbf{Key Components}:
        \begin{enumerate}
            \item \textbf{iOS Client App}: Live camera feed and intelligent edge-streaming.
            \item \textbf{Backend Server (Raspberry Pi)}: High-efficiency ALPR edge-node processing.
            \item \textbf{Verification Logic}: Real-time regex matching and database lookup.
        \end{enumerate}
    \end{itemize}
\end{frame}

\begin{frame}{1. AI Model Optimization for the Edge}
    \begin{itemize}
        \item \textbf{Model}: Ultralytics YOLOv11.
        \item \textbf{Challenge}: Heavy FP32 floating-point calculations are too slow for an ARM CPU.
        \item \textbf{Solution}: Static Quantization using ONNX Runtime.
        \begin{itemize}
            \item Reduced model footprint by $\approx 75\%$.
            \item Shifted INT8 operator weights to UINT8 via direct graph manipulation to ensure hardware compatibility with the target runtime.
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{2. Backend Server Architecture}
    \begin{itemize}
        \item \textbf{Platform}: Raspberry Pi running a Python Asyncio WebSocket Server.
        \item \textbf{Workflow}:
        \begin{enumerate}
            \item Receives JPEG-compressed frames from the iOS client.
            \item Runs YOLOv11 ONNX inference natively.
            \item Crops detected regions and passes them to \texttt{EasyOCR}.
            \item Validates extracted text against standard license plate regexes (e.g., \texttt{MH12DE1433}).
            \item Queries the local database for the vehicle owner and account balance.
            \item Responds with precise JSON payloads (detections or final verification).
        \end{enumerate}
    \end{itemize}
\end{frame}

\begin{frame}{3. Native iOS Client App}
    \begin{itemize}
        \item \textbf{Architecture}: Built entirely in Swift/SwiftUI with no third-party ML dependencies on the client-side.
        \item \textbf{Features}:
        \begin{itemize}
            \item \textbf{Camera Manager}: Leverages \texttt{AVFoundation} to capture and compress high-resolution frames dynamically.
            \item \textbf{Live Overlay}: Synthesizes server coordinate responses (bounding boxes + OCR text) over the live camera preview.
            \item \textbf{Autonomous Halt}: Automatically detects a successful verification JSON payload, halts the camera session, and displays a clean ``Access Granted'' modal.
        \end{itemize}
    \end{itemize}
\end{frame}
